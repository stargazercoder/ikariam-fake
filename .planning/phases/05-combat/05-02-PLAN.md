```dart
class BattleRepository {
  const BattleRepository();

  /// Stream battles where current user is the attacker.
  Stream<List<Battle>> watchBattlesAsAttacker(String userId) {
    return Supabase.instance.client
      .from('battles')
      .stream(primaryKey: ['id'])
      .eq('attacker_id', userId)
      .map((rows) => rows.map(Battle.fromJson).toList());
  }

  /// Stream battles where current user is the defender.
  Stream<List<Battle>> watchBattlesAsDefender(String userId) {
    return Supabase.instance.client
      .from('battles')
      .stream(primaryKey: ['id'])
      .eq('defender_id', userId)
      .map((rows) => rows.map(Battle.fromJson).toList());
  }

  /// Stream turn-by-turn results for a specific battle.
  Stream<List<BattleTurn>> watchBattleTurns(String battleId) {
    return Supabase.instance.client
      .from('battle_turns')
      .stream(primaryKey: ['id'])
      .eq('battle_id', battleId)
      .order('turn_number')
      .map((rows) => rows.map(BattleTurn.fromJson).toList());
  }
}

final battleRepositoryProvider = Provider<BattleRepository>((ref) => const BattleRepository());
```

Note: Two separate streams needed because Supabase Realtime .stream() does not support OR filters (established decision from Phase 4).

**lib/features/battles/providers/battles_provider.dart:**

- `allMyBattlesProvider` — Provider.autoDispose that watches both attacker and defender providers using the current user ID from auth state. Merges and deduplicates by battle ID, sorts by createdAt descending (newest first). Uses the pattern from 05-RESEARCH.md code examples (dual-stream merge with seen set).
